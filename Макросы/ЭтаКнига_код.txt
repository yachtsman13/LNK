Option Explicit

Private Sub Workbook_Open()
    On Error Resume Next
    Application.DisplayAlerts = True

    ' Книга данных открывается сразу: без неё не работает ни один пункт меню.
    If Данные() Is Nothing Then Exit Sub

    ' Рабочее место должно быть подписано, иначе ID записей на разных
    ' машинах совпадут и при слиянии одна затрёт другую.
    If КодРабочегоМеста() = "" Then
        ПроверитьКодРабочегоМеста
    End If

    НапомнитьПроСрокиИМастера
End Sub


Private Sub Workbook_BeforeClose(Cancel As Boolean)
    On Error Resume Next
    ' Резервная копия, уровень 1: одна в день, тихо.
    modBackup.СделатьЛокальнуюКопию True
    If Not Данные() Is Nothing Then Данные().Close SaveChanges:=True
End Sub


' Короткая сводка при запуске: просроченное и кто сегодня мастер.
Private Sub НапомнитьПроСрокиИМастера()
    Dim текст As String
    Dim мастер As String

    текст = ПроверитьПоверкуСИ(Date)

    мастер = ПрочитатьМастера()
    If мастер <> "" Then текст = текст & IIf(текст = "", "", vbCrLf & vbCrLf) & мастер

    If текст <> "" Then MsgBox текст, vbInformation, "ЛНК"
End Sub


Private Function ПрочитатьМастера() As String
    Dim файл As String
    Dim f As Integer
    Dim s As String

    файл = ПапкаОбмена() & "Свод" & Application.PathSeparator & "мастер.txt"
    If Dir(файл) = "" Then Exit Function
    On Error Resume Next
    f = FreeFile
    Open файл For Input As #f
    If Not EOF(f) Then Line Input #f, s
    Close #f
    If Trim$(s) <> "" Then ПрочитатьМастера = "Мастер сегодня: " & s
End Function
