Option Explicit

Private Sub Workbook_Open()
    On Error Resume Next
    Application.DisplayAlerts = True

    ' Книга данных открывается сразу: без неё не работает ни один пункт меню.
    If Данные() Is Nothing Then Exit Sub

    ' Рабочее место должно быть подписано, иначе ID записей на разных
    ' машинах совпадут и при слиянии одна затрёт другую.
    If КодРабочегоМеста() = "" Then
        ПроверитьКодРабочегоМеста
    End If

    ' Дашборд строится при каждом открытии: сроки меняются каждый день,
    ' а вчерашняя картинка хуже, чем никакой.
    modDashboard.ПостроитьДашборд
End Sub


Private Sub Workbook_BeforeClose(Cancel As Boolean)
    On Error Resume Next
    ' Резервная копия, уровень 1: одна в день, тихо.
    modBackup.СделатьЛокальнуюКопию True
    If Not Данные() Is Nothing Then Данные().Close SaveChanges:=True
End Sub
