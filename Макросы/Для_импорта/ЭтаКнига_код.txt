' =======================================================================
'  Код модуля «ЭтаКнига» (ThisWorkbook) — не отдельного модуля!
'
'  Куда вставлять: Alt+F11 -> VBAProject (ЛНК_Программа.xlsm)
'  -> Microsoft Excel Objects -> ЭтаКнига -> двойной щелчок
'  -> вставить в открывшееся окно. Установщик делает это сам.
'
'  ПОЧЕМУ ЗДЕСЬ ВЕЗДЕ Application.Run, А НЕ ПРЯМЫЕ ВЫЗОВЫ МОДУЛЕЙ.
'
'  Этот код попадает в книгу ОТДЕЛЬНО от модулей: текстом, а не импортом.
'  Значит, возможно состояние, когда он в книге уже есть, а модулей ещё
'  нет — установку не довели до конца, или код вставили руками в чистую
'  книгу. Раньше отсюда модули звались напрямую (modMenu.ПостроитьПульт),
'  и в таком состоянии Excel выдавал «Compile error: Variable not
'  defined» с подсветкой имени модуля: при Option Explicit неизвестное
'  имя модуля читается как необъявленная переменная.
'
'  Беда не в самом сообщении, а в том, что оно значит: проект VBA
'  собирается ЦЕЛИКОМ или не собирается вовсе. Одна такая строка — и не
'  работает ничего: ни лента, ни пульт, ни правая кнопка мыши. Причём
'  причина («не установлены модули») из сообщения никак не следует, а
'  вкладка на ленте при этом на месте — она лежит в файле книги и от
'  кода не зависит. Снаружи это выглядит как «программа не работает и не
'  говорит почему».
'
'  Application.Run ищет имя во время выполнения. Проект собирается
'  всегда, а чего не хватает — сказано словами и с указанием, что делать.
' =======================================================================
Option Explicit

' Чего не хватило при открытии. Собирается по ходу, показывается один
' раз в конце: пятнадцать одинаковых окон подряд никто не читает.
Private mНеНайдено As String


Private Sub Workbook_Open()
    Dim данные As Object

    Application.DisplayAlerts = True
    mНеНайдено = ""

    ' Книга данных открывается сразу: без неё не работает ни один пункт меню.
    Set данные = ПозватьЗаКнигой("modConfig.Данные")
    If данные Is Nothing Then
        СообщитьОНедостающем
        Exit Sub
    End If

    ' Рабочее место должно быть подписано, иначе ID записей на разных
    ' машинах совпадут и при слиянии одна затрёт другую.
    If ПозватьЗаСтрокой("modConfig.КодРабочегоМеста") = "" Then
        Позвать "modConfig.ПроверитьКодРабочегоМеста"
    End If

    ' Пульт строится при каждом открытии: сроки меняются каждый день,
    ' а вчерашняя картинка хуже, чем никакой. Заодно пересоздаются
    ' кнопки — книгу могли скопировать с машины, где лист пульта
    ' остался от прежней версии.
    Позвать "modMenu.ПостроитьПульт"

    ' Подменю «ЛНК» в правой кнопке мыши. Собирается каждый раз заново:
    ' пункты добавляются Temporary, Excel их не сохраняет, а после
    ' аварийного закрытия старые могли остаться.
    Позвать "modUI.ПостроитьМенюПоПравойКнопке"

    ' Незакрытый лист ввода замеров мог остаться с прошлого раза: до
    ' записи в «Замеры» этих данных нет больше нигде, поэтому лист
    ' переживает закрытие книги, а здесь о нём напоминают.
    Позвать "modEntry.ПродолжитьНезакрытыйВвод"

    ' Модули и книга данных должны быть одной версии — иначе ошибка
    ' вылезет позже и совсем не там, где на самом деле дело.
    Позвать "modUpdate.СверитьВерсиюКода"

    ПозватьС "modUpdate.ПроверитьОбновление", True

    СообщитьОНедостающем
End Sub


Private Sub Workbook_BeforeClose(Cancel As Boolean)
    Dim данные As Object

    mНеНайдено = ""

    ' Меню правой кнопки убирается первым: если книгу закрыли, а Excel
    ' остался открытым с чужим файлом, подменю «ЛНК» в нём только мешает
    ' — команды всё равно уже некому выполнять.
    Позвать "modUI.УбратьМенюПоПравойКнопке"

    ' Резервная копия, уровень 1: одна в день, тихо.
    ПозватьС "modBackup.СделатьЛокальнуюКопию", True

    Set данные = ПозватьЗаКнигой("modConfig.Данные")
    If Not данные Is Nothing Then
        On Error Resume Next
        данные.Close SaveChanges:=True
        On Error GoTo 0
    End If
End Sub


' -- Вызов модулей -----------------------------------------------------

' Ошибка 1004 у Application.Run значит одно: такой процедуры в книге нет.
' Это не сбой внутри неё, а именно отсутствие модуля — и различать эти
' два случая обязательно, иначе «модули не установлены» будет
' показываться по любому поводу.
Private Const НЕТ_ТАКОЙ As Long = 1004


Private Sub Позвать(имя As String)
    On Error Resume Next
    Application.Run имя
    Запомнить имя
    On Error GoTo 0
End Sub


Private Sub ПозватьС(имя As String, аргумент As Variant)
    On Error Resume Next
    Application.Run имя, аргумент
    Запомнить имя
    On Error GoTo 0
End Sub


' Для функций, у которых важен ответ. Их две, и возвращают они разное —
' объект и строку, — поэтому и помощника два. Объединять их в один с
' проверкой типа нельзя: проверить тип, не позвав, невозможно, а звать
' дважды — значит дважды выполнить процедуру.
Private Function ПозватьЗаКнигой(имя As String) As Object
    On Error Resume Next
    Set ПозватьЗаКнигой = Application.Run(имя)
    Запомнить имя
    On Error GoTo 0
End Function


Private Function ПозватьЗаСтрокой(имя As String) As String
    On Error Resume Next
    ПозватьЗаСтрокой = CStr(Application.Run(имя))
    Запомнить имя
    On Error GoTo 0
End Function


' Err читается сразу: любое обращение к On Error его очищает.
Private Sub Запомнить(имя As String)
    Dim номер As Long
    Dim модуль As String

    номер = Err.Number
    If номер = 0 Then Exit Sub
    Err.Clear
    If номер <> НЕТ_ТАКОЙ Then Exit Sub

    модуль = Left$(имя, InStr(имя & ".", ".") - 1)
    If InStr(mНеНайдено & ", ", модуль & ",") = 0 Then
        mНеНайдено = mНеНайдено & IIf(mНеНайдено = "", "", ", ") & модуль
    End If
End Sub


Private Sub СообщитьОНедостающем()
    If mНеНайдено = "" Then Exit Sub

    MsgBox "Модули системы в этой книге установлены не полностью." & vbCrLf & vbCrLf & _
           "Не найдено: " & mНеНайдено & vbCrLf & vbCrLf & _
           "Что сделать: Alt+F11 -> в дереве слева модуль «Установка» -> " & _
           "поставить в него курсор -> F5 -> УстановитьВсё -> указать папку " & _
           "«Для_импорта» из присланного архива." & vbCrLf & vbCrLf & _
           "Если модуля «Установка» в дереве нет — импортируйте его: " & _
           "File -> Import File… -> «Установка.bas» из той же папки.", _
           vbExclamation, "Система ЛНК"
    mНеНайдено = ""
End Sub
