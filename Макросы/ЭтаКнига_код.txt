Option Explicit

Private Sub Workbook_Open()
    On Error Resume Next
    Application.DisplayAlerts = True

    ' Книга данных открывается сразу: без неё не работает ни один пункт меню.
    If Данные() Is Nothing Then Exit Sub

    ' Рабочее место должно быть подписано, иначе ID записей на разных
    ' машинах совпадут и при слиянии одна затрёт другую.
    If КодРабочегоМеста() = "" Then
        ПроверитьКодРабочегоМеста
    End If

    ' Пульт строится при каждом открытии: сроки меняются каждый день,
    ' а вчерашняя картинка хуже, чем никакой. Заодно пересоздаются
    ' кнопки — книгу могли скопировать с машины, где лист пульта
    ' остался от прежней версии.
    ' On Error Resume Next стоит выше: если пульт или проверка
    ' обновления почему-то не отработают, книга всё равно откроется.
    modMenu.ПостроитьПульт

    ' Подменю «ЛНК» в правой кнопке мыши. Собирается каждый раз заново:
    ' пункты добавляются Temporary, Excel их не сохраняет, а после
    ' аварийного закрытия старые могли остаться.
    modUI.ПостроитьМенюПоПравойКнопке

    modUpdate.ПроверитьОбновление True
End Sub


Private Sub Workbook_BeforeClose(Cancel As Boolean)
    On Error Resume Next
    ' Меню правой кнопки убирается первым: если книгу закрыли, а Excel
    ' остался открытым с чужим файлом, подменю «ЛНК» в нём только мешает
    ' — команды всё равно уже некому выполнять.
    modUI.УбратьМенюПоПравойКнопке

    ' Резервная копия, уровень 1: одна в день, тихо.
    modBackup.СделатьЛокальнуюКопию True
    If Not Данные() Is Nothing Then Данные().Close SaveChanges:=True
End Sub
